function calculateSalary() { loadDataFromStorage(); const month = document.getElementById('salMonth').value; const empId = document.getElementById('salEmpSelect').value; let daysWorkedVal = parseFloat(document.getElementById('daysWorked').value); if (isNaN(daysWorkedVal) || daysWorkedVal < 0) daysWorkedVal = 0; if (!month || !empId) { resetPayslip(); return; } const empObj = employees.find(e => String(e.id || e.empId || e.employeeId).trim() === String(empId).trim()); if (!empObj) return; const { dailyBasic, dailyEpf, allowances } = getEmployeeSalaryDetails(empObj); const key = `${empId}_${month}`; const monthAttendance = attendanceData[key] || {}; let totalOtHours = 0; let totalAdvance = 0; let totalLoan = 0; Object.values(monthAttendance).forEach(day => { totalOtHours += parseFloat(day.ot || day.otHours || 0); totalAdvance += parseFloat(day.advance || 0); totalLoan += parseFloat(day.loan || 0); }); // 1. Basic Salary Calculation (Daily Basic * Days Worked) const workedBasicSalary = dailyBasic * daysWorkedVal; // 2. OT Calculation (Daily Basic / 8 * OT Hours) const otRate = dailyBasic / 8; const otEarnings = otRate * totalOtHours; // 3. EPF & ETF Base (Daily EPF Amount * Days Worked) const epfApplicableAmount = dailyEpf * daysWorkedVal; const epf8 = epfApplicableAmount * 0.08; const epf12 = epfApplicableAmount * 0.12; const etf3 = epfApplicableAmount * 0.03; // 4. Gross Earnings & Net Deductions const grossEarnings = workedBasicSalary + otEarnings + allowances; const totalDeductions = epf8 + totalAdvance + totalLoan; const netSalary = grossEarnings - totalDeductions; // Displaying Results in Payslip UI document.getElementById('dispMonth').innerText = `Month: ${month}`; document.getElementById('dispEmpId').innerText = empObj.id || empObj.empId || empObj.employeeId; document.getElementById('dispEmpName').innerText = empObj.name; document.getElementById('dispEmpBranch').innerText = empObj.branch; document.getElementById('dispDaysWorked').innerText = `${daysWorkedVal} Days`; document.getElementById('lblDailyBasic').innerText = formatLKR(dailyBasic); document.getElementById('lblDaysWorked').innerText = daysWorkedVal; document.getElementById('dispWorkedBasic').innerText = formatLKR(workedBasicSalary); document.getElementById('dispOtHours').innerText = totalOtHours.toFixed(1); document.getElementById('dispOtRate').innerText = formatLKR(otRate); document.getElementById('dispOtAmount').innerText = formatLKR(otEarnings); document.getElementById('dispAllowances').innerText = formatLKR(allowances); document.getElementById('dispGrossEarnings').innerText = formatLKR(grossEarnings); document.getElementById('dispEpfBase').innerText = formatLKR(epfApplicableAmount); document.getElementById('dispEpf8').innerText = formatLKR(epf8); document.getElementById('dispAdvance').innerText = formatLKR(totalAdvance); document.getElementById('dispLoan').innerText = formatLKR(totalLoan); document.getElementById('dispTotalDeductions').innerText = formatLKR(totalDeductions); document.getElementById('dispNetSalary').innerText = `${formatLKR(netSalary)} LKR`; document.getElementById('dispEpf12').innerText = `${formatLKR(epf12)} LKR`; document.getElementById('dispEtf3').innerText = `${formatLKR(etf3)} LKR`; }
